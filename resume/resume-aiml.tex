\section{Selected AI/ML Projects}

\projectEntryPaper{MorphoCLIP}{PyTorch, DINOv3, BioClinical ModernBERT, contrastive learning, CPJUMP1}
{https://github.com/suxrobgm/morphoclip}
{https://arxiv.org/abs/2608.22690}
\begin{itemize}
    \item Matches Cell Painting microscopy of cells treated with drugs or genetic perturbations to plain-language descriptions of the treatment, using contrastive image-text training.
    \item The DINOv3 and BioClinical ModernBERT backbones stay frozen and only the projection heads train, so it fits on one consumer GPU.
    \item Evaluated on CPJUMP1 (51 plates, 3M+ images). As first author I led design, training, and evaluation.
\end{itemize}

\projectEntry{Localize, Don't Beautify}{Image-editing APIs, inpainting, ArcFace, prompt engineering}
{https://arxiv.org/abs/2608.02841}
\begin{itemize}
    \item Commercial image editors asked for one surgical change tend to retouch the whole face. I tested prompt steering, masked compositing, and inpainting as client-side fixes across six editors on 196 preview edits.
    \item Scored identity preservation with ArcFace and checked whether each edit stayed in its region. Plain masked compositing kept edits in place best. Sole author.
\end{itemize}

\projectEntryPaper{MelanomaNet}{PyTorch, EfficientNet V2, focal loss, GradCAM++, OpenCV, ISIC 2019}
{https://github.com/suxrobgm/explainable-melanoma}
{https://arxiv.org/abs/2512.09289}
\begin{itemize}
    \item Nine-class skin-lesion classifier on ISIC 2019 (25K images). EfficientNet V2 with focal loss to handle the class imbalance; 0.86 weighted F1.
    \item Scores the model's GradCAM++ attention against the ABCDE criteria dermatologists use, each measured from the lesion mask. Sole author.
\end{itemize}

\projectEntry{Med Image Scanner}{FastAPI, PyTorch, OpenCV, OHIF Viewer, DICOM, Next.js}
{https://github.com/suxrobgm/med-image-scanner}
\begin{itemize}
    \item Pulls X-ray, CT, and MRI studies from hospital imaging systems over DICOM and runs PyTorch detectors on them, such as pneumonia on chest X-rays and hemorrhage on head CTs.
    \item Results show up as overlays in OHIF. Studies are de-identified on the fly, and access is role-based and logged.
\end{itemize}
