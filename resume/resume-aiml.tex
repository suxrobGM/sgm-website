\section{Selected AI/ML Projects}

\projectEntryPaper{MorphoCLIP}{PyTorch, DINOv3, BioClinical ModernBERT, contrastive learning, CPJUMP1}
{https://github.com/suxrobgm/morphoclip}
{https://arxiv.org/abs/2608.22690}
\begin{itemize}
    \item Built a text-supervised contrastive model that matches Cell Painting microscopy images of drug- and gene-perturbed cells to descriptions of the treatment in ordinary language. The backbones (DINOv3, BioClinical ModernBERT) stay frozen and only small projection heads train, so it fits on one consumer GPU. Tested on CPJUMP1: 51 plates, 3M+ images. Published on arXiv, joint work with two Northeastern co-authors; I led the design, training, and evaluation.
\end{itemize}

\projectEntry{Localize, Don't Beautify}{Image-editing APIs, inpainting, ArcFace, prompt engineering}
{https://arxiv.org/abs/2608.02841}
\begin{itemize}
    \item Ask a commercial image editor to change one feature of a face and it beautifies the whole thing, a problem when the picture is meant to be a surgical preview. Tested how much of that can be fixed client-side, without touching the model: prompt-only steering, masked compositing, and model-based inpainting across six editors. Plain masking localized better. Published on arXiv.
\end{itemize}

\projectEntryPaper{MelanomaNet}{PyTorch, EfficientNet V2, GradCAM++, OpenCV}
{https://github.com/suxrobgm/explainable-melanoma}
{https://arxiv.org/abs/2512.09289}
\begin{itemize}
    \item Trained a deep-learning model to classify skin lesions across nine diagnostic categories on a standard public dermatology dataset of 25K+ images. It reaches 0.86 weighted F1.
    \item The interesting part is the interpretability. The model shows heatmaps of where it looked, broken down along the same ABCDE criteria (Asymmetry, Border, Color, Diameter, Evolution) dermatologists already work from, so a clinician can check the reasoning. Published on arXiv.
\end{itemize}

\projectEntry{Med Image Scanner}{FastAPI, PyTorch, OpenCV, OHIF Viewer, DICOM, Next.js}
{https://github.com/suxrobgm/med-image-scanner}
\begin{itemize}
    \item HIPAA-compliant platform that pulls X-ray, CT, and MRI scans straight from hospital imaging systems and runs PyTorch detection models over them. It flags pneumonia on chest X-rays and intracranial hemorrhage on head CTs.
    \item Predictions render as overlays inside OHIF, the viewer radiologists already work in. De-identification happens on the fly, with audit logging and role-based access behind it.
\end{itemize}

\projectEntryDual{LogisticsX}{Claude API, MCP, .NET 10, EF Core, SignalR, Angular 21, Kotlin KMP, PostgreSQL}
